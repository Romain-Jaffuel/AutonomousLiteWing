# Python 3.11 only: the litewing library pins requires-python >=3.11,<3.12.
#
# Rebuild the environment from scratch:
#
#   uv venv --python 3.11
#   uv pip install --python .venv --system-certs \
#       "cflib @ git+https://github.com/Circuit-Digest/crazyflie-lib-python@0fb27688e39c665b4f01fe8d84330959cb44bdf4"
#   uv pip install --python .venv --system-certs matplotlib==3.11.2
#   uv pip install --python .venv --system-certs --no-deps \
#       "litewing @ git+https://github.com/Circuit-Digest/LiteWing-Library@949a7c9d1ae14cd26e62f574e3fbfd896c5cd7e1"
#
# Two things this order works around:
#
# --system-certs is required on this machine. TLS is intercepted here, so the
# bundled root certificates fail with UnknownIssuer. Git needs the same fix:
# git config --global http.sslBackend schannel
#
# --no-deps on litewing is deliberate. Its own metadata depends on
# "cflib @ .../archive/refs/heads/master.zip", an unpinned branch archive, so
# installing it normally silently moves cflib under us. cflib is pinned above
# by commit instead.

cflib @ git+https://github.com/Circuit-Digest/crazyflie-lib-python@0fb27688e39c665b4f01fe8d84330959cb44bdf4
litewing @ git+https://github.com/Circuit-Digest/LiteWing-Library@949a7c9d1ae14cd26e62f574e3fbfd896c5cd7e1
matplotlib==3.11.2
